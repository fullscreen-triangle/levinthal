\begin{figure}
\caption{\textbf{Panel 1: Desaturation two-step HAT kinetics.}
Left: individual rate constants for the two HAT steps and radical rebound.
Center: effective desaturation rate as a function of $k_2$, showing that
rebound competition reduces $k_\text{desat,eff}$ well below $k_1$.
Right: 3D surface of $k_\text{eff}$ as a function of both $k_1$ and $k_2$.}
\label{fig:desat_twostep}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 2: Kinetic isotope effect for desaturation.}
Left: comparison of intrinsic KIE for single-HAT vs.\ desaturation.
Center: effective KIE as a function of $\Delta M$ for the first HAT step,
showing attenuation from commitment to the second step.
Right: KIE surface as a function of C--H stretching frequency and temperature.}
\label{fig:desat_kie}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 3: Arene oxide formation.}
Left: comparison of epoxidation and aliphatic HAT rate constants.
Center: rate vs.\ $\Delta M$ with vertical markers at the two mechanisms.
Right: ratio surface $k_\text{epox}/k_\text{HAT}$ over $\Delta M$ parameter space.}
\label{fig:arene_oxide}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 4: NIH shift kinetics.}
Left: NIH shift vs.\ epoxidation rate bars.
Center: horizontal bar chart of all five atypical reaction rates.
Right: 3D visualization of the NIH shift rate over a spherical geometry.}
\label{fig:nih_shift}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 5: Nucleophilic O-atom transfer.}
Left: barrier height as a function of $\Delta M$ with the nucleophilic
$\Delta M = 0.42$ marked.
Center: comparison of nucleophilic, epoxidation, and effective desaturation rates.
Right: averaged rate surface over two-parameter $\Delta M$ space.}
\label{fig:nucleophilic}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 6: Rate ordering of all five atypical mechanisms.}
Left: horizontal bar chart ordered from slowest (desaturation) to fastest
(NIH shift).
Center: scatter plot of $\Delta M$ vs.\ rate showing monotonic relationship.
Right: 3D phase-space representation of rate and $\Delta M$ for all five modes.}
\label{fig:rate_ordering}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 7: Product partitioning after arene oxide formation.}
Left: pie chart of phenol vs.\ trans-dihydrodiol fractions under
non-enzymatic conditions.
Center: phenol fraction as a function of competing hydration rate.
Right: 3D surface of phenol fraction over $k_\text{NIH}$--$k_\text{hydration}$
parameter space.}
\label{fig:partitioning}
\end{figure}

\begin{figure}
\caption{\textbf{Panel 8: Validation summary.}
Left: predicted rate constants (dots) overlaid on literature ranges (bars)
for all five reaction classes; all predictions fall within the literature
range (green dots).
Center: KIE values; desaturation is the only mechanism with KIE $> 1$.
Right: 3D scatter of all five mechanisms in $\Delta M$--rate--KIE space.}
