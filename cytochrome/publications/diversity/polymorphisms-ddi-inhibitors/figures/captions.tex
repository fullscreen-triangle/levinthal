\begin{figure*}[!htbp]

\includegraphics[width=\textwidth]{figures/panel_03_mbi_kinetics.png}
\caption{\textbf{Mechanism-based inactivation kinetics for three CYP3A4 inactivators.}
(\textbf{A}) Time-dependent enzyme activity $E(t)/E_0 = \exp(-(k_\text{obs}+k_\text{deg})t)$
for clarithromycin ($K_I = 3.7$\,$\mu$M, $k_\text{inact} = 0.040$\,min$^{-1}$),
diltiazem ($K_I = 14.0$\,$\mu$M, $k_\text{inact} = 0.060$\,min$^{-1}$), and
erythromycin ($K_I = 72.0$\,$\mu$M, $k_\text{inact} = 0.025$\,min$^{-1}$) at
clinical unbound concentrations; clarithromycin reduces CYP3A4 to $< 50$\,\%
within 60\,min, consistent with its classification as a strong mechanism-based
inactivator.
(\textbf{B}) Kitz--Wilson surface of observed inactivation rate $k_\text{obs}([I], K_I)$;
the potency ratio $k_\text{inact}/K_I$ (clarithromycin 0.011, diltiazem 0.004,
erythromycin 0.0003\,min$^{-1}\mu$M$^{-1}$) governs the rate of active-enzyme
depletion at clinical concentrations.}
\label{fig:panel03_mbi}

\caption{\textbf{CYP3A4 transcriptional induction and victim drug AUC reduction.}
(\textbf{A}) Victim drug AUC ratio $R_\text{AUC} = 1/E_\text{fold}$ versus
fold-induction for rifampicin ($E_\text{fold} = 20$, $R_\text{AUC} = 0.05$),
phenobarbital ($E_\text{fold} = 10$, $R_\text{AUC} = 0.10$), and omeprazole
($E_\text{fold} = 5$, $R_\text{AUC} = 0.20$); each inducer reduces victim-drug
exposure below the $< 10$\,\% clinical significance threshold.
(\textbf{B}) PXR $E_\text{max}$ model for rifampicin:
$E_\text{fold}([R]) = 1 + 30\cdot[R]/(0.50 + [R])$ predicting $\approx 24$-fold
induction at 2\,$\mu$M, consistent with the 20-fold midazolam AUC decrease
reported by Backman et al.\ and encoded as $\Delta M_\text{induced} = \Delta M_0 - \ln E_\text{fold}$.}

\end{figure*}
